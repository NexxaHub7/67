local function parseMoney(s)
    if not s or s == "" then return 0 end
    s = s:gsub("[%$,%s]", "")
    local n, suf = s:match("^([%d%.]+)([KMBTQ]?)")
    if not n then return 0 end
    n = tonumber(n) or 0
    local m = ({K = 1e3, M = 1e6, B = 1e9, T = 1e12, Q = 1e15})[suf] or 1
    return n * m
end

local function formatGen(n)
    if not n or n == 0 then return "0/s" end
    if n >= 1e15 then return string.format("%.2fQ/s", n / 1e15) end
    if n >= 1e12 then return string.format("%.2fT/s", n / 1e12) end
    if n >= 1e9 then return string.format("%.2fB/s", n / 1e9) end
    if n >= 1e6 then return string.format("%.2fM/s", n / 1e6) end
    if n >= 1e3 then return string.format("%.2fK/s", n / 1e3) end
    return tostring(math.floor(n)) .. "/s"
end

local function readOverhead(o)
    if not o then return nil end
    local function t(ns)
        for _, n in ipairs(ns) do
            local l = o:FindFirstChild(n)
            if l and l.Text and l.Text ~= "" then return l.Text end
        end
        return ""
    end
    return {
        displayName = t({"DisplayName", "Name", "Title", "Label"}),
        mutation = t({"Mutation", "Mut", "Mutacion"}),
        rarity = t({"Rarity", "Rare", "Rareza"}),
        generation = t({"Generation", "Gen", "Income", "Money", "CPS", "PerSecond"}),
        price = t({"Price", "Value", "Precio", "Cost"}),
        stolen = t({"Stolen", "Stole", "Robado", "Status"}),
    }
end

local function getPodiumPrompt(p)
    local b = p:FindFirstChild("Base")
    if not b then return nil end
    local s = b:FindFirstChild("Spawn")
    if not s then return nil end
    local a = s:FindFirstChild("PromptAttachment")
    if not a then return nil end
    return a:FindFirstChild("ProximityPrompt")
end

local function isStealablePrompt(p)
    if not p or not p.Parent then return false end
    if not p.Enabled then return false end
    local st = p:GetAttribute("State")
    if st == "Steal" or st == "Grab" then return true end
    local at = p.ActionText
    if at == "Steal" or at == "Grab" then return true end
    if at:lower():find("steal") then return true end
    if at:lower():find("rob") then return true end
    return false
end

local function scanAllClowns()
    local ps = Workspace:FindFirstChild("Plots")
    if not ps then return end
    local all = {}
    local best = nil
    local bv = -1
    for _, p in ipairs(ps:GetChildren()) do
        if p.Name ~= STATE.MY_UUID then
            local pod = p:FindFirstChild("AnimalPodiums")
            if pod then
                for _, po in ipairs(pod:GetChildren()) do
                    local b = po:FindFirstChild("Base")
                    if b then
                        local oh = b:FindFirstChild("AnimalOverhead", true)
                        if not oh then
                            for _, d in ipairs(b:GetDescendants()) do
                                if d:IsA("BillboardGui") then oh = d break end
                            end
                        end
                        local data = readOverhead(oh)
                        if data and data.displayName ~= "" and data.displayName ~= "Empty" then
                            local st = data.stolen:upper()
                            if st ~= "IN FUSE" and st ~= "FUSE" and st ~= "FUSING" then
                                local pr = getPodiumPrompt(po)
                                if not pr then
                                    for _, d in ipairs(b:GetDescendants()) do
                                        if d:IsA("ProximityPrompt") then pr = d break end
                                    end
                                end
                                if pr and isStealablePrompt(pr) then
                                    local g = parseMoney(data.generation)
                                    if g == 0 and data.price ~= "" then g = parseMoney(data.price) end
                                    local i = {
                                        plot = p.Name,
                                        slot = po.Name,
                                        nombre = data.displayName,
                                        mutacion = data.mutation,
                                        gen = g,
                                        total = g,
                                        price = parseMoney(data.price),
                                        rarity = data.rarity,
                                        prompt = pr,
                                        base = b,
                                    }
                                    table.insert(all, i)
                                    if g > bv then bv = g best = i end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    table.sort(all, function(a, b) return a.total > b.total end)
    STATE.TOP_CLOWNS = all
    if best then STATE.BEST_CLOWN = best end
end

local function tpToClown(c, dg)
    if not c then return false, "Sin datos" end
    local ps = Workspace:FindFirstChild("Plots")
    local p = ps and ps:FindFirstChild(c.plot)
    local pod = p and p:FindFirstChild("AnimalPodiums")
    local po = pod and pod:FindFirstChild(tostring(c.slot))
    local b = po and po:FindFirstChild("Base")
    if not b then return false, "Sin base" end
    local pp = b.PrimaryPart or b:FindFirstChildWhichIsA("BasePart", true)
    if not pp then return false, "Sin part" end
    local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if h then
        h.CFrame = pp.CFrame + Vector3.new(0, 3, 0)
        h.AssemblyLinearVelocity = Vector3.zero
        if dg and STATE.AUTO_GRAB_ON then
            task.wait(CONFIG.AutoGrabDelay)
            autoGrabAt(c)
        end
        return true, c.nombre
    end
    return false, "Sin HRP"
end

local function cleanRagdoll(c)
    if not c then return end
    c:SetAttribute("Ragdolled", nil)
    c:SetAttribute("__RealRagdoll", nil)
    c:SetAttribute("__RD_RequiresNeck", nil)
    c:SetAttribute("__RD_AutoRotate", nil)
    local rt = c:FindFirstChild("__RagdollRuntime")
    if rt then rt:Destroy() end
    for _, d in ipairs(c:GetDescendants()) do
        if d:IsA("BasePart") then
            local oc = d:GetAttribute("__RD_OldCollide")
            if oc ~= nil then d.CanCollide = oc d:SetAttribute("__RD_OldCollide", nil) end
            local om = d:GetAttribute("__RD_OldMassless")
            if om ~= nil then d.Massless = om d:SetAttribute("__RD_OldMassless", nil) end
        elseif d:IsA("Motor6D") then
            local oe = d:GetAttribute("__RD_OldEnabled")
            if oe ~= nil then d.Enabled = oe d:SetAttribute("__RD_OldEnabled", nil) end
        end
    end
end

local function startAntiRagdoll()
    if antiRagdollRunning then return end
    antiRagdollRunning = true
    local function hook(ch)
        if not ch then return end
        cleanRagdoll(ch)
        local cn
        cn = RunService.Heartbeat:Connect(function()
            if not STATE.ANTI_RAGDOLL_ON then return end
            if ch.Parent == nil then if cn then cn:Disconnect() end return end
            if ch:GetAttribute("Ragdolled") then cleanRagdoll(ch) end
        end)
        table.insert(antiRagdollConns, cn)
    end
    if LocalPlayer.Character then hook(LocalPlayer.Character) end
    table.insert(antiRagdollConns, LocalPlayer.CharacterAdded:Connect(function(c)
        task.wait(0.2)
        hook(c)
    end))
end

local function stopAntiRagdoll()
    STATE.ANTI_RAGDOLL_ON = false
    for _, c in ipairs(antiRagdollConns) do pcall(function() c:Disconnect() end) end
    antiRagdollConns = {}
    antiRagdollRunning = false
end

local function setAntiRagdoll(on)
    STATE.ANTI_RAGDOLL_ON = on
    if on then startAntiRagdoll() else stopAntiRagdoll() end
end

local function getPayasoName()
    local i = LocalPlayer:GetAttribute("StealingIndex")
    if i == nil then return "?" end
    local ps = Workspace:FindFirstChild("Plots")
    if not ps then return tostring(i) end
    for _, p in ipairs(ps:GetChildren()) do
        local pod = p:FindFirstChild("AnimalPodiums")
        if pod then
            local po = pod:FindFirstChild(tostring(i))
            if po then
                local b = po:FindFirstChild("Base")
                if b then
                    local oh = b:FindFirstChild("AnimalOverhead", true)
                    if not oh then
                        for _, d in ipairs(b:GetDescendants()) do
                            if d:IsA("BillboardGui") then oh = d break end
                        end
                    end
                    local data = readOverhead(oh)

                    end

        end
    end
    return tostring(i)
end

local function esperarStealingFalse(t)
    t = t or CONFIG.StealTimeout
    local t0 = os.clock()
    while LocalPlayer:GetAttribute("Stealing") == true do
        if os.clock() - t0 > t then break end
        task.wait(0.05)
    end
end

local function onStealStart()
    if STATE.STEAL_ROUTINE_RUNNING then return end
    STATE.STEAL_ROUTINE_RUNNING = true
    local ch = LocalPlayer.Character
    local h = ch and ch:FindFirstChild("HumanoidRootPart")
    local ocf = h and h.CFrame or nil
    task.wait(0.1)
    local py = getPayasoName()
    print("[AutoTP] Payaso robado:", py)
    if STATE.REVENGE_ON and h then
        local cl, di = nil, 25
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local th = p.Character:FindFirstChild("HumanoidRootPart")
                if th then
                    local d = (th.Position - h.Position).Magnitude
                    if d < di then di = d cl = p end
                end
            end
        end
        if cl then
            STATE.REVENGE_TARGET = cl
            STATE.REVENGE_TIME = os.clock()
            if _G.BrainrotNotify then _G.BrainrotNotify("Revenge", "Te esta robando: " .. cl.Name) end
        end
    end
    pcall(doTPBase)
    task.spawn(function()
        esperarStealingFalse(CONFIG.StealTimeout)
        pcall(function()
            local c = LocalPlayer.Character
            local hh = c and c:FindFirstChild("HumanoidRootPart")
            if hh and ocf then
                hh.CFrame = ocf
                hh.AssemblyLinearVelocity = Vector3.zero
            end
        end)
        if STATE.KICK_ON_STEAL then
            task.wait(0.2)
            local r = "Payaso Robado: " .. tostring(py) .. "."
            if _G.BrainrotShowReason then _G.BrainrotShowReason(r) end
            task.wait(1)
            LocalPlayer:Kick(r)
        end
        STATE.STEAL_ROUTINE_RUNNING = false
    end)
end

local function findSnipeTarget()
    local ps = Workspace:FindFirstChild("Plots")
    if not ps then return nil end
    for _, p in ipairs(ps:GetChildren()) do
        if p.Name ~= STATE.MY_UUID then
            local pod = p:FindFirstChild("AnimalPodiums")
            if pod then
                for _, po in ipairs(pod:GetChildren()) do
                    local b = po:FindFirstChild("Base")
                    if b then
                        local oh = b:FindFirstChild("AnimalOverhead", true)
                        if not oh then
                            for _, d in ipairs(b:GetDescendants()) do
                                if d:IsA("BillboardGui") then oh = d break end
                            end
                        end
                        local data = readOverhead(oh)
                        if data and data.displayName ~= "" and data.displayName ~= "Empty" then
                            local r = tostring(data.rarity)
                            if STATE.SNIPE_SELECTED[r] then
                                local st = data.stolen:upper()
                                if st ~= "IN FUSE" and st ~= "FUSE" and st ~= "FUSING" then
                                    local pr = getPodiumPrompt(po)
                                    if not pr then
                                        for _, d in ipairs(b:GetDescendants()) do
                                            if d:IsA("ProximityPrompt") then pr = d break end
                                        end
                                    end
                                    if pr and isStealablePrompt(pr) then
                                        return {
                                            plot = p.Name,
                                            slot = po.Name,
                                            nombre = data.displayName,
                                            rarity = r,
                                            gen = parseMoney(data.generation),
                                            prompt = pr,
                                            base = b,
                                        }
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return nil
end

local function runSnipe()
    if STATE.SNIPE_RUNNING then return end
    if os.clock() - STATE.SNIPE_LAST < CONFIG.SnipeCooldown then return end
    local t = findSnipeTarget()
    if not t then return end
    STATE.SNIPE_RUNNING = true
    STATE.SNIPE_LAST = os.clock()
    print("[Snipe] Encontrado:", t.nombre, "|", t.rarity)
    pcall(function() tpToClown(t, true) end)
    if not STATE.INSTANT_STEAL_ON then
        pcall(function() LocalPlayer:SetAttribute("InstantSteal", true) end)
    end
    task.delay(2, function() STATE.SNIPE_RUNNING = false end)
end

local function esTrabado()
    local c = LocalPlayer.Character
    if not c then return false end
    local h = c:FindFirstChild("HumanoidRootPart")
    local hu = c:FindFirstChildOfClass("Humanoid")
    if not h or not hu then return false end
    if h.Anchored then return true end
    if hu.MoveDirection.Magnitude > 0.1 and h.AssemblyLinearVelocity.Magnitude < 0.5 then return true end
    for _, w in ipairs(h:GetChildren()) do
        if w:IsA("WeldConstraint") or w:IsA("Weld") or w:IsA("Motor6D") then
            local o = (w.Part0 == h) and w.Part1 or w.Part0
            if o and not o:IsDescendantOf(c) then return true end
        end
    end
    return false
end

local function destrabar()
    local c = LocalPlayer.Character
    if not c then return end
    local h = c:FindFirstChild("HumanoidRootPart")
    local hu = c:FindFirstChildOfClass("Humanoid")
    if not h then return end
    pcall(function() h.Anchored = false end)
    if hu then
        pcall(function()
            if hu.WalkSpeed < 16 then hu.WalkSpeed = 16 end
            if hu.JumpPower < 50 then hu.JumpPower = 50 end
        end)
    end
    for _, w in ipairs(h:GetChildren()) do
        if w:IsA("WeldConstraint") or w:IsA("Weld") or w:IsA("Motor6D") then
            local o = (w.Part0 == h) and w.Part1 or w.Part0
            if o and not o:IsDescendantOf(c) then pcall(function() w:Destroy() end) end
        end
    end
    for _, p in ipairs(c:GetDescendants()) do
        if p:IsA("BasePart") and p.Anchored then pcall(function() p.Anchored = false end) end
    end
end

local function findThiefNearMyPlot()
    if not STATE.MY_UUID then return nil end
    local ps = Workspace:FindFirstChild("Plots")
    local mp = ps and ps:FindFirstChild(STATE.MY_UUID)
    if not mp then return nil end
    local ms = mp:FindFirstChild("Spawn") or mp.PrimaryPart
    if not ms then return nil end
    local mpos = ms.Position
    local cl, cd = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local h = p.Character:FindFirstChild("HumanoidRootPart")
            local hu = p.Character:FindFirstChildOfClass("Humanoid")
            if h and hu and hu.Health > 0 then
                local d = (h.Position - mpos).Magnitude
                if d < CONFIG.BaseDetectRadius and d < cd then cd = d cl = p end
            end
        end
    end
    return cl
end

local function findPlayerPlotByUsername(user)
    local ps = Workspace:FindFirstChild("Plots")
    if not ps then return nil end
    for _, plot in ipairs(ps:GetChildren()) do
        local sign = plot:FindFirstChild("PlotSign")
        if sign then
            local nm = sign:FindFirstChild("Name", true)
                or sign:FindFirstChild("Username", true)
                or sign:FindFirstChild("DisplayName", true)
            if nm and nm.Text and nm.Text:lower():find(user:lower(), 1, true) then
                return plot
            end
        end
        local podiums = plot:FindFirstChild("AnimalPodiums")
        if podiums then
            for _, podium in ipairs(podiums:GetChildren()) do
                local b = podium:FindFirstChild("Base")
                if b then
                    local oh = b:FindFirstChild("AnimalOverhead", true)
                    local data = oh and readOverhead(oh)
                    if data and data.displayName and data.displayName:lower():find(user:lower(), 1, true) then
                        return plot
                    end
                end
            end
        end
    end
    return nil
end

local function findMyEmptySlots()
    local ps = Workspace:FindFirstChild("Plots")
    if not ps then return 0 end
    local my = ps:FindFirstChild(STATE.MY_UUID)
    if not my then return 0 end
    local podiums = my:FindFirstChild("AnimalPodiums")
    if not podiums then return 0 end
    local n = 0
    for _, podium in ipairs(podiums:GetChildren()) do
        local b = podium:FindFirstChild("Base")
        if b then
            local oh = b:FindFirstChild("AnimalOverhead", true)
            local data = oh and readOverhead(oh)
            if not data or data.displayName == "" or data.displayName == "Empty" then
                n = n + 1
            end
        end
    end
    return n
end

local function collectStealableFromPlot(plot)
    local list = {}
    if not plot then return list end
    local podiums = plot:FindFirstChild("AnimalPodiums")
    if not podiums then return list end
    for _, podium in ipairs(podiums:GetChildren()) do
        local b = podium:FindFirstChild("Base")
        if b then
            local oh = b:FindFirstChild("AnimalOverhead", true)
            local data = oh and readOverhead(oh)
            if data and data.displayName ~= "" and data.displayName ~= "Empty" then
                local st = data.stolen:upper()
                if st ~= "IN FUSE" and st ~= "FUSE" and st ~= "FUSING" then
                    local pr = getPodiumPrompt(podium)
                    if not pr then
                        for _, d in ipairs(b:GetDescendants()) do
                            if d:IsA("ProximityPrompt") then pr = d break end
                        end
                    end
                    if pr and isStealablePrompt(pr) then
                        table.insert(list, {
                            plot = plot.Name,
                            slot = podium.Name,
                            nombre = data.displayName,
                            rarity = data.rarity,
                            gen = parseMoney(data.generation),
                            prompt = pr,
                            base = b,
                        })
                    end
                end
            end
        end
    end
    return list
end

local function emptyDatabase()
    if STATE.EMPTYING then return end
    if not STATE.EMPTY_TARGET_USER or STATE.EMPTY_TARGET_USER == "" then
        if _G.BrainrotNotify then
            _G.BrainrotNotify("Empty DB", "Primero ingresa un usuario y dale Save")
        end
        return
    end
    STATE.EMPTYING = true
    task.spawn(function()
        while STATE.EMPTYING do
            if findMyEmptySlots() <= 0 then
                if _G.BrainrotNotify then
                    _G.BrainrotNotify("Empty DB", "Tu base esta llena")
                end
                STATE.EMPTYING = false
                break
            end
            local plot = findPlayerPlotByUsername(STATE.EMPTY_TARGET_USER)
            if not plot then
                if _G.BrainrotNotify then
                    _G.BrainrotNotify("Empty DB", "No encontre al usuario")
                end
                STATE.EMPTYING = false
                break
            end
            local list = collectStealableFromPlot(plot)
            if #list == 0 then
                if _G.BrainrotNotify then
                    _G.BrainrotNotify("Empty DB", "Base vaciada")
                end
                STATE.EMPTYING = false
                break
            end
            table.sort(list, function(a, b)
                return (a.gen or 0) > (b.gen or 0)
            end)
            local obj = list[1]
            pcall(function()
                tpToClown(obj, true)
            end)
            if not STATE.INSTANT_STEAL_ON then
                pcall(function()
                    LocalPlayer:SetAttribute("InstantSteal", true)
                end)
            end
            task.wait(1.2)
        end
    end)
end

local function makeDraggable(frame, handle)
    handle = handle or frame
    local dragging, start, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            start = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - start
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

    end)
    btn.MouseLeave:Connect(function()
        tween(btn, TI_Fast, {BackgroundTransparency = btn:GetAttribute("BaseTrans") or 0})
    end)
end

local pages, tabButtons = {}, {}
local currentTab = nil

local function showTab(name)
    for n, page in pairs(pages) do
        if n == name then
            page.Visible = true
            page.Position = UDim2.fromOffset(8, 14)
            tween(page, TI_Med, {Position = UDim2.fromOffset(8, 8)})
        else
            page.Visible = false
        end
    end
    for n, btn in pairs(tabButtons) do
        if n == name then
            tween(btn, TI_Fast, {BackgroundColor3 = Theme.Accent})
            btn.TextColor3 = Theme.Text
        else
            tween(btn, TI_Fast, {BackgroundColor3 = Theme.ToggleOff})
            btn.TextColor3 = Theme.TextDim
        end
    end
    currentTab = name
end

local function slider(page, title, minV, maxV, default, callback)
    local row = Instance.new("Frame")
    row.Size = UDim2.new(1, 0, 0, 54)
    row.BackgroundColor3 = Theme.Bg
    row.BorderSizePixel = 0
    row.Parent = page
    corner(row, 9)

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(0.65, 0, 0, 20)
    lbl.Position = UDim2.fromOffset(12, 6)
    lbl.BackgroundTransparency = 1
    lbl.Text = title
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 13
    lbl.TextColor3 = Theme.Text
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = row

    local valLbl = Instance.new("TextLabel")
    valLbl.Size = UDim2.new(0.35, -12, 0, 20)
    valLbl.Position = UDim2.new(0.65, 0, 0, 6)
    valLbl.BackgroundTransparency = 1
    valLbl.Text = tostring(default)
    valLbl.Font = Enum.Font.GothamBold
    valLbl.TextSize = 13
    valLbl.TextColor3 = Theme.Accent2
    valLbl.TextXAlignment = Enum.TextXAlignment.Right
    valLbl.Parent = row

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -24, 0, 6)
    track.Position = UDim2.new(0, 12, 0, 36)
    track.BackgroundColor3 = Theme.ToggleOff
    track.BorderSizePixel = 0
    track.Parent = row
    corner(track, 3)

    local fill = Instance.new("Frame")
    local pct = (default - minV) / (maxV - minV)
    fill.Size = UDim2.new(pct, 0, 1, 0)
    fill.BackgroundColor3 = Theme.Accent
    fill.BorderSizePixel = 0
    fill.Parent = track
    corner(fill, 3)
    gradient(fill, Theme.Accent, Theme.Accent2, 0)

    local value, sliding = default, false
    local function setFromX(x)
        local rel = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        value = math.floor(minV + rel * (maxV - minV) + 0.5)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        valLbl.Text = tostring(value)
        if callback then callback(value) end
    end
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = true
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = false
        end
    end)
end

local uiOpen = true

local function openUI()
    uiOpen = true
    OpenBtn.Visible = false
    Main.Visible = true
    Main.Size = UDim2.fromOffset(WIN_W, 0)
    Body.Visible = true
    tween(Main, TI_Spring, {Size = UDim2.fromOffset(WIN_W, WIN_H)})

local function minimizeUI()
    uiOpen = false
    savedPos = Main.Position
    local tw = tween(Main, TI_Med, {Size = UDim2.fromOffset(WIN_W, 0)})
    tw.Completed:Connect(function()
        Main.Visible = false
        OpenBtn.Visible = true
        OpenBtn.Size = UDim2.fromOffset(0, 0)
        tween(OpenBtn, TI_Spring, {Size = UDim2.fromOffset(38, 38)})
    end)
end

local function closeUI()
    uiOpen = false
    local tw = tween(Main, TI_Med, {
        Size = UDim2.fromOffset(0, 0),
        Position = UDim2.new(Main.Position.X.Scale, Main.Position.X.Offset + WIN_W/2,
                             Main.Position.Y.Scale, Main.Position.Y.Offset + WIN_H/2),
    })
    tw.Completed:Connect(function()
        Main.Visible = false
        Main.Position = savedPos
        OpenBtn.Visible = true
        OpenBtn.Size = UDim2.fromOffset(0, 0)
        tween(OpenBtn, TI_Spring, {Size = UDim2.fromOffset(38, 38)})
    end)
end

    end
end)

end)
